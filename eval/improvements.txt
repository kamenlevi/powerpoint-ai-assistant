To handle requests for adding a hyperlink to specific text on a slide without overwriting existing text, find the shape with the matching text and set the hyperlink.
```javascript
if (prompt.includes("Add a hyperlink to the text")) {
  const text = prompt.match(/'([^']+)'/)[1];
  const url = prompt.match(/https?:\/\/([^']+)/)[1];
  const slide = await getSlideByIndex(0);
  for (const shape of slide.shapes.items) {
    try {
      const textRange = shape.textFrame.textRange;
      if (textRange.text.includes(text)) {
        textRange.hyperlink.address = url;
        await context.sync();
      }
    } catch (e) {}
  }
}
```

To handle requests for deleting a specific slide, check the slide index and provide a message if it does not exist.
```javascript
if (prompt.includes("delete the second slide")) {
  const index = 1;
  const slides = await listSlides();
  if (index >= 0 && index < slides.length) {
    const slide = await getSlideByIndex(index);
    slide.delete();
    await context.sync();
  } else {
    console.log(`Slide ${index + 1} does not exist.`);
  }
}
```
