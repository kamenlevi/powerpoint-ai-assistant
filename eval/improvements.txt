To handle requests for adding a hyperlink to specific text without overwriting existing text, find the shape with the matching text and apply the hyperlink.
```javascript
if (prompt.includes("add a hyperlink to the text") && prompt.includes("that links to")) {
  const slideIndex = 0;
  const slide = await getSlideByIndex(slideIndex);
  const textToLink = prompt.match(/add a hyperlink to the text '(.+)'/)[1];
  const link = prompt.match(/that links to (.+)/)[1];
  slide.shapes.load("items/textFrame/textRange");
  await context.sync();
  const shape = slide.shapes.items.find(s => s.textFrame && s.textFrame.text.includes(textToLink));
  if (shape) {
    const textRange = shape.textFrame.getTextRange();
    const foundText = textRange.text.includes(textToLink);
    if (foundText) {
      textRange.hyperlink.address = link;
    } else {
      console.log("Text not found.");
    }
    await context.sync();
  }
}
```

To handle requests for deleting a slide, use the slide.delete method and check if the slide exists.
```javascript
if (prompt.includes("delete the")) {
  const slideNumber = prompt.match(/delete the (\d+)/)[1];
  const slideIndex = slideNumber - 1;
  try {
    const slide = await getSlideByIndex(slideIndex);
    if (slide) {
      slide.delete();
      await context.sync();
    } else {
      console.log("Slide not found.");
    }
  } catch (e) {
    console.log("Error deleting slide.");
  }
}
```
