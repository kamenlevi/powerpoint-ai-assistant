To handle requests for adding a slide with a specific font, use the shape.textFrame.textRange.font.name property.
```javascript
if (prompt.includes("make the body text on the")) {
  const font = prompt.match(/make the body text on the (.+) slide use (.+) font/)[2];
  const slideNumber = prompt.match(/make the body text on the (.+) slide/)[1];
  const slideIndex = slideNumber === "first" ? 0 : parseInt(slideNumber) - 1;
  const slide = await getSlideByIndex(slideIndex);
  slide.shapes.load("items/placeholderFormat/type, items/textFrame/textRange/font");
  await context.sync();
  const body = slide.shapes.items.find(s => s.placeholderFormat.type === "body");
  if (body) {
    body.textFrame.textRange.font.name = font;
  }
  await context.sync();
}
```

  await context.sync();

    title.textFrame.textRange.font.size = parseInt(size);

  await context.sync();
}
