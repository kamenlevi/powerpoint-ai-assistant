To handle requests for changing a slide's title, load the slide's shapes and set the title text.
```javascript
// Change a slide's title
if (prompt.includes("change slide")) {
  const index = parseInt(prompt.match(/slide (\d+)/)[1]) - 1;
  const title = prompt.match(/'([^']+)'/)[1];
  const slide = await getSlideByIndex(index);
  slide.shapes.load("items/placeholderFormat/type, items/textFrame/textRange");
  await context.sync();
  const titleShape = slide.shapes.items.find(s => s.placeholderFormat.type === "title");
  if (!titleShape) throw new Error(`No title shape on slide ${index + 1}.`);
  titleShape.textFrame.textRange.text = title;
  await context.sync();
}
```

// Move a slide

  await moveSlide(index, position);
}

To handle requests for deleting a slide by index, check if the slide index is valid and use the delete method.
```javascript
// Delete a slide by index
if (prompt.includes("delete slide")) {
  const index = parseInt(prompt.match(/slide (\d+)/)[1]) - 1;
  const slides = await listSlides();
  if (index >= 0 && index < slides.length) {
    const slide = await getSlideByIndex(index);
    slide.delete();
    await context.sync();
  } else {
    console.log(`Slide ${index + 1} does not exist.`);
  }
}
```
