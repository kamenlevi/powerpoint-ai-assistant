To handle requests for changing the text color of a specific title, use the shape.textFrame.textRange.font.color property.
```javascript
if (prompt.includes("make the title of slide")) {
  const color = prompt.match(/make the title of slide (.+) (#.+)$/)[2];
  const slideNumber = prompt.match(/make the title of slide (.+) (#.+)$/)[1];
  const slideIndex = slideNumber === "1"? 0 : parseInt(slideNumber) - 1;
  const slide = await getSlideByIndex(slideIndex);
  slide.shapes.load("items/placeholderFormat/type, items/textFrame/textRange/font");
  await context.sync();
  const title = slide.shapes.items.find(s => s.placeholderFormat.type === "title");
  if (title) {
    title.textFrame.textRange.font.color = color;
  }
  await context.sync();
}
```

To handle requests for replacing text across the deck, use the presentation.slides.load method and iterate over each slide and shape.
```javascript
if (prompt.includes("replace all instances of")) {
  const oldText = prompt.match(/replace all instances of (.+) with/)[1];
  const newText = prompt.match(/replace all instances of.+ with (.+)/)[1];
  presentation.slides.load("items/shapes/items/textFrame/textRange/text");
  await context.sync();
  for (const slide of presentation.slides.items) {
    for (const shape of slide.shapes.items) {
      if (shape.textFrame) {
        shape.textFrame.textRange.text = shape.textFrame.textRange.text.replace(new RegExp(oldText, 'g'), newText);
      }
    }
  }
  await context.sync();
}
```
